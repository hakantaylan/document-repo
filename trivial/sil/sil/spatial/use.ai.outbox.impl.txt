3. Transaction-Scoped Event Collector
java


@Component
@Scope(value = ConfigurableBeanFactory.SCOPE_SINGLETON)
public class TransactionEventCollector {
    
    private static final ThreadLocal<TransactionEventContext> TRANSACTION_CONTEXT = 
        new ThreadLocal<>();
    
    public void addEvent(DomainEvent event, String aggregateType, String aggregateId) {
        TransactionEventContext context = getOrCreateContext();
        context.addEvent(event, aggregateType, aggregateId);
    }
    
    public void clearContext() {
        TRANSACTION_CONTEXT.remove();
    }
    
    public TransactionEventContext getContext() {
        return TRANSACTION_CONTEXT.get();
    }
    
    private TransactionEventContext getOrCreateContext() {
        TransactionEventContext context = TRANSACTION_CONTEXT.get();
        if (context == null) {
            context = new TransactionEventContext();
            TRANSACTION_CONTEXT.set(context);
        }
        return context;
    }
    
    @Data
    public static class TransactionEventContext {
        private final String transactionId = UUID.randomUUID().toString();
        private final Map<String, AggregateEvents> eventsByAggregate = new HashMap<>();
        
        public void addEvent(DomainEvent event, String aggregateType, String aggregateId) {
            String key = aggregateType + ":" + aggregateId;
            eventsByAggregate.computeIfAbsent(key, 
                k -> new AggregateEvents(aggregateType, aggregateId))
                .addEvent(event);
        }
        
        public Collection<AggregateEvents> getAllAggregateEvents() {
            return eventsByAggregate.values();
        }
    }
    
    @Data
    public static class AggregateEvents {
        private final String aggregateType;
        private final String aggregateId;
        private final List<DomainEvent> events = new ArrayList<>();
        
        public void addEvent(DomainEvent event) {
            events.add(event);
        }
    }
}
4. Transparent Domain Event Publisher
java


@Component
@Transactional
public class OutboxDomainEventPublisher implements DomainEventPublisher {
    
    private final TransactionEventCollector eventCollector;
    private final ObjectMapper objectMapper;
    
    public OutboxDomainEventPublisher(TransactionEventCollector eventCollector,
                                    ObjectMapper objectMapper) {
        this.eventCollector = eventCollector;
        this.objectMapper = objectMapper;
    }
    
    @Override
    public void publishEvent(DomainEvent event) {
        String aggregateInfo = extractAggregateInfo(event);
        String[] parts = aggregateInfo.split(":");
        eventCollector.addEvent(event, parts[0], parts[1]);
    }
    
    @Override
    public void publishEvents(Collection<DomainEvent> events) {
        events.forEach(this::publishEvent);
    }
    
    private String extractAggregateInfo(DomainEvent event) {
        // Extract aggregate type and ID from event
        // This can be done via reflection, annotations, or interface methods
        if (event instanceof AggregateAwareEvent awareEvent) {
            return awareEvent.getAggregateType() + ":" + awareEvent.getAggregateId();
        }
        
        // Fallback to reflection-based extraction
        return extractViaReflection(event);
    }
    
    private String extractViaReflection(DomainEvent event) {
        // Implementation depends on your aggregate identification strategy
        // Could use annotations like @AggregateType, @AggregateId
        // or naming conventions
        return "Unknown:Unknown";
    }
}
public interface AggregateAwareEvent {
    String getAggregateType();
    String getAggregateId();
}
5. Transaction Synchronization for Outbox Persistence
java


@Component
public class OutboxTransactionSynchronization implements TransactionSynchronization {
    
    private final TransactionEventCollector eventCollector;
    private final OutboxEventRepository outboxRepository;
    private final ObjectMapper objectMapper;
    
    public OutboxTransactionSynchronization(TransactionEventCollector eventCollector,
                                          OutboxEventRepository outboxRepository,
                                          ObjectMapper objectMapper) {
        this.eventCollector = eventCollector;
        this.outboxRepository = outboxRepository;
        this.objectMapper = objectMapper;
    }
    
    @Override
    public void beforeCommit(boolean readOnly) {
        if (readOnly) return;
        
        TransactionEventCollector.TransactionEventContext context = 
            eventCollector.getContext();
        
        if (context != null && !context.getAllAggregateEvents().isEmpty()) {
            persistOutboxEvents(context);
        }
    }
    
    @Override
    public void afterCompletion(int status) {
        eventCollector.clearContext();
    }
    
    private void persistOutboxEvents(TransactionEventCollector.TransactionEventContext context) {
        try {
            for (var aggregateEvents : context.getAllAggregateEvents()) {
                OutboxEvent outboxEvent = createOutboxEvent(context, aggregateEvents);
                outboxRepository.save(outboxEvent);
            }
        } catch (Exception e) {
            throw new OutboxPersistenceException("Failed to persist outbox events", e);
        }
    }
    
    private OutboxEvent createOutboxEvent(
            TransactionEventCollector.TransactionEventContext context,
            TransactionEventCollector.AggregateEvents aggregateEvents) throws Exception {
        
        String eventsJson = objectMapper.writeValueAsString(aggregateEvents.getEvents());
        
        OutboxEvent outboxEvent = new OutboxEvent();
        outboxEvent.setTransactionId(context.getTransactionId());
        outboxEvent.setAggregateType(aggregateEvents.getAggregateType());
        outboxEvent.setAggregateId(aggregateEvents.getAggregateId());
        outboxEvent.setEvents(eventsJson);
        outboxEvent.setCreatedAt(LocalDateTime.now());
        
        return outboxEvent;
    }
}

7. Aspect for Automatic Registration
java


@Aspect
@Component
public class TransactionOutboxAspect {
    
    private final OutboxTransactionSynchronization synchronization;
    
    public TransactionOutboxAspect(OutboxTransactionSynchronization synchronization) {
        this.synchronization = synchronization;
    }
    
    @Around("@annotation(org.springframework.transaction.annotation.Transactional)")
    public Object registerOutboxSynchronization(ProceedingJoinPoint joinPoint) throws Throwable {
        boolean registeredHere = false;
        
        if (TransactionSynchronizationManager.isSynchronizationActive() && 
            !isAlreadyRegistered()) {
            TransactionSynchronizationManager.registerSynchronization(synchronization);
            registeredHere = true;
        }
        
        try {
            return joinPoint.proceed();
        } catch (Exception e) {
            throw e;
        }
    }
    
    private boolean isAlreadyRegistered() {
        return TransactionSynchronizationManager.getSynchronizations()
            .stream()
            .anyMatch(sync -> sync instanceof OutboxTransactionSynchronization);
    }
}

@Repository
public interface OutboxEventRepository extends JpaRepository<OutboxEvent, UUID> {
    
    @Query("SELECT o FROM OutboxEvent o WHERE o.status = :status ORDER BY o.createdAt ASC")
    List<OutboxEvent> findByStatusOrderByCreatedAtAsc(@Param("status") OutboxStatus status);
    
    @Modifying
    @Query("UPDATE OutboxEvent o SET o.status = :status, o.processedAt = :processedAt WHERE o.id = :id")
    void updateStatus(@Param("id") UUID id, 
                     @Param("status") OutboxStatus status, 
                     @Param("processedAt") LocalDateTime processedAt);
}
@Component
public class OutboxEventProcessor {
    
    private final OutboxEventRepository repository;
    private final ApplicationEventPublisher eventPublisher;
    private final ObjectMapper objectMapper;
    
    @Scheduled(fixedDelay = 5000)
    @Transactional
    public void processOutboxEvents() {
        List<OutboxEvent> pendingEvents = repository.findByStatusOrderByCreatedAtAsc(
            OutboxStatus.PENDING);
        
        for (OutboxEvent outboxEvent : pendingEvents) {
            try {
                processOutboxEvent(outboxEvent);
                markAsProcessed(outboxEvent);
            } catch (Exception e) {
                markAsFailed(outboxEvent);
                // Log error
            }
        }
    }
    
    private void processOutboxEvent(OutboxEvent outboxEvent) throws Exception {
        List<DomainEvent> events = objectMapper.readValue(
            outboxEvent.getEvents(), 
            new TypeReference<List<DomainEvent>>() {}
        );
        
        events.forEach(eventPublisher::publishEvent);
    }
    
    private void markAsProcessed(OutboxEvent outboxEvent) {
        repository.updateStatus(outboxEvent.getId(), 
                              OutboxStatus.PROCESSED, 
                              LocalDateTime.now());
    }
    
    private void markAsFailed(OutboxEvent outboxEvent) {
        repository.updateStatus(outboxEvent.getId(), 
                              OutboxStatus.FAILED, 
                              LocalDateTime.now());
    }
}

Key Features of This Infrastructure:
Transparent: Developers just inject DomainEventPublisher and publish events normally
Transaction Boundary Respect: Uses TransactionSynchronization to ensure events are persisted only on successful commit
Single Row per Transaction: Groups all events from the same transaction into one outbox entry
Automatic Registration: Aspects automatically register transaction synchronization
Type Safety: Uses proper Java 25 features like pattern matching and records where applicable
Hibernate 7.2.1 Compatible: Uses modern JPA features and JSON support
Spring Boot 4 Ready: Leverages auto-configuration and modern Spring features
This infrastructure ensures that all domain events within a transaction boundary are reliably captured and stored as a single outbox entry, making the outbox pattern completely transparent to your business logic code
